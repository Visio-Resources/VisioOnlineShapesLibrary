Create a .env file in this folder:

FLASK_APP=app
FLASK_ENV=development
SECRET_KEY=change_me
DATABASE_URI=sqlite:///app.db
MAIL_SERVER=smtp.mymailserver.com
MAIL_PORT=587
MAIL_USE_TLS=True
MAIL_USE_SSL=False
MAIL_USERNAME=mail@mymailserver.com
MAIL_PASSWORD=my_email_password
MAIL_DEFAULT_SENDER="My Name <mail@mymailserver.com>"
